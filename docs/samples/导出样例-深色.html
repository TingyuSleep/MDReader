export function slugify(text: string): string {
  return text
    .trim()
    .toLowerCase()
    .replace(/[^\p{L}\p{N}\s-]/gu, "")
    .replace(/\s+/g, "-");
}

def count_words(text: str) -> int:
    """中日韩逐字 + 西文按词"""
    return len([c for c in text if c.strip()])
